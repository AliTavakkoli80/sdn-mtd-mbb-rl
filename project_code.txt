class FlowManager:
    def __init__(self, app):
        self.app = app

    def add_flow(self, datapath, priority, match, actions, buffer_id=None,
                 idle_timeout=0, hard_timeout=0, table_id=0):
        ofproto = datapath.ofproto
        parser = datapath.ofproto_parser
        inst = [parser.OFPInstructionActions(ofproto.OFPIT_APPLY_ACTIONS, actions)]
        if buffer_id and buffer_id != ofproto.OFP_NO_BUFFER:
            mod = parser.OFPFlowMod(
                datapath=datapath, buffer_id=buffer_id, priority=priority,
                match=match, instructions=inst, table_id=table_id,
                idle_timeout=idle_timeout, hard_timeout=hard_timeout)
        else:
            mod = parser.OFPFlowMod(
                datapath=datapath, priority=priority, match=match,
                instructions=inst, table_id=table_id,
                idle_timeout=idle_timeout, hard_timeout=hard_timeout)
        datapath.send_msg(mod)
        self.app.logger.debug(f"[flow] installed on S{datapath.id}: prio={priority} match={match}")

    def install_table_miss(self, datapath):
        ofproto = datapath.ofproto
        parser = datapath.ofproto_parser
        match = parser.OFPMatch()
        actions = [parser.OFPActionOutput(ofproto.OFPP_CONTROLLER, ofproto.OFPCML_NO_BUFFER)]
        self.add_flow(datapath, const.PRIORITY_TABLE_MISS, match, actions)

    def install_translation_flows(self, ingress_dp, in_port, src_ip, virtual_dst, real_dst):
        log = self.app.logger
        if real_dst not in self.app.topology.host_table:
            log.error(f"[flow] Destination {real_dst} unknown - abort")
            return False

        dst_info = self.app.topology.host_table[real_dst]
        egress_dpid = dst_info['dpid']
        egress_port = dst_info['port']
        virtual_src = self.app.real_to_virtual.get(src_ip, src_ip)
        parser = ingress_dp.ofproto_parser
        ofproto = ingress_dp.ofproto

        # Ingress: virtual→real + NORMAL
        if ingress_dp.id in self.app.topology.access_ports:
            match = parser.OFPMatch(eth_type=const.ETH_TYPE_IP, ipv4_dst=virtual_dst)
            actions = [parser.OFPActionSetField(ipv4_dst=real_dst),
                       parser.OFPActionOutput(ofproto.OFPP_NORMAL)]
            self.add_flow(ingress_dp, const.PRIORITY_TRANSLATION, match, actions,
                          idle_timeout=const.TRANSLATION_FLOW_IDLE_TIMEOUT)

        # Egress (if different): real→virtual + direct output
        if egress_dpid != ingress_dp.id:
            egress_dp = self.app.topology.get_datapath(egress_dpid)
            if egress_dp and egress_dpid in self.app.topology.access_ports:
                match = parser.OFPMatch(eth_type=const.ETH_TYPE_IP, ipv4_src=src_ip, ipv4_dst=real_dst)
                actions = [parser.OFPActionSetField(ipv4_src=virtual_src),
                           parser.OFPActionOutput(egress_port)]
                self.add_flow(egress_dp, const.PRIORITY_TRANSLATION, match, actions,
                              idle_timeout=const.TRANSLATION_FLOW_IDLE_TIMEOUT)

                # Reverse path: from egress to ingress
                match_rev = parser.OFPMatch(eth_type=const.ETH_TYPE_IP, ipv4_dst=virtual_src)
                actions_rev = [parser.OFPActionSetField(ipv4_dst=src_ip),
                               parser.OFPActionOutput(ofproto.OFPP_NORMAL)]
                self.add_flow(egress_dp, const.PRIORITY_TRANSLATION, match_rev, actions_rev,
                              idle_timeout=const.TRANSLATION_FLOW_IDLE_TIMEOUT)

        # Reverse path at ingress: from dest to source
        if ingress_dp.id in self.app.topology.access_ports:
            match_rev_src = parser.OFPMatch(eth_type=const.ETH_TYPE_IP, ipv4_src=real_dst, ipv4_dst=src_ip)
            actions_rev_src = [parser.OFPActionSetField(ipv4_src=virtual_dst),
                               parser.OFPActionOutput(in_port)]
            self.add_flow(ingress_dp, const.PRIORITY_TRANSLATION, match_rev_src, actions_rev_src,
                          idle_timeout=const.TRANSLATION_FLOW_IDLE_TIMEOUT)

        log.info("[flow] ✅ Translation flows installed")
        return True

    def install_normal_flow(self, datapath, src_ip, dst_ip, in_port, buffer_id, data):
        parser = datapath.ofproto_parser
        ofproto = datapath.ofproto
        actions = [parser.OFPActionOutput(ofproto.OFPP_NORMAL)]
        match = parser.OFPMatch(eth_type=const.ETH_TYPE_IP, ipv4_src=src_ip, ipv4_dst=dst_ip)
        self.add_flow(datapath, const.PRIORITY_NORMAL, match, actions,
                      idle_timeout=const.NORMAL_FLOW_IDLE_TIMEOUT,
                      hard_timeout=const.NORMAL_FLOW_HARD_TIMEOUT)
        if buffer_id != ofproto.OFP_NO_BUFFER:
            out = parser.OFPPacketOut(datapath=datapath, buffer_id=buffer_id,
                                      in_port=in_port, actions=actions)
        else:
            out = parser.OFPPacketOut(datapath=datapath, buffer_id=ofproto.OFP_NO_BUFFER,
                                      in_port=in_port, actions=actions, data=data)
        datapath.send_msg(out)
        self.app.logger.debug(f"[normal] flow for {src_ip}→{dst_ip}")

from ryu.lib.packet import packet, ethernet, ipv4, udp
from . import constants as const
from dnslib import DNSRecord, QTYPE, A

class IPv4Handler:
    def __init__(self, app):
        self.app = app

    def handle_ipv4(self, datapath, in_port, pkt, eth, ipv4_pkt, buffer_id, raw_data):
        src, dst = ipv4_pkt.src, ipv4_pkt.dst
        proto = {1:'ICMP',6:'TCP',17:'UDP'}.get(ipv4_pkt.proto, str(ipv4_pkt.proto))
        self.app.logger.info(f"[ipv4] {src} → {dst} | {proto} | TTL={ipv4_pkt.ttl}")
        self.app.topology.register_host(datapath.id, in_port, src, eth.src)

        # DNS response
        if ipv4_pkt.proto == const.IP_PROTO_UDP:
            udp_pkt = pkt.get_protocol(udp.udp)
            if udp_pkt and udp_pkt.src_port == const.DNS_SERVER_PORT:
                self.app.logger.info("[dns] DNS response intercepted")
                self._handle_dns_response(datapath, in_port, pkt, eth, ipv4_pkt, udp_pkt, buffer_id, raw_data)
                return

        # MTD translation
        if dst in self.app.virtual_to_real:
            real_dst = self.app.virtual_to_real[dst]
            self.app.logger.info(f"[mtd] virtual {dst} → real {real_dst}")
            self.app.flow_manager.install_translation_flows(datapath, in_port, src, dst, real_dst)
            self._translate_and_forward(datapath, in_port, dst, real_dst, buffer_id, raw_data)
        else:
            self.app.logger.info("[ipv4] normal traffic")
            self.app.flow_manager.install_normal_flow(datapath, src, dst, in_port, buffer_id, raw_data)

    def _translate_and_forward(self, datapath, in_port, virtual_dst, real_dst, buffer_id, raw_data):
        parser = datapath.ofproto_parser
        ofproto = datapath.ofproto
        actions = [
            parser.OFPActionSetField(ipv4_dst=real_dst),
            parser.OFPActionOutput(ofproto.OFPP_NORMAL)
        ]
        out = parser.OFPPacketOut(
            datapath=datapath,
            buffer_id=buffer_id,
            in_port=in_port,
            actions=actions,
            data=raw_data if buffer_id == ofproto.OFP_NO_BUFFER else None
        )
        datapath.send_msg(out)
        self.app.logger.info("[mtd] first packet translated")

    def _handle_dns_response(self, datapath, in_port, pkt, eth, ipv4_pkt, udp_pkt, buffer_id, raw_data):
        try:
            ip_hdr_len = ipv4_pkt.header_length * 4
            dns_start = 14 + ip_hdr_len + 8
            if len(raw_data) < dns_start + 12:
                self._forward_original(datapath, in_port, buffer_id, raw_data)
                return

            dns_payload = raw_data[dns_start:]
            dns_resp = DNSRecord.parse(dns_payload)

            if not (dns_resp.header.qr == 1 and dns_resp.rr):
                self._forward_original(datapath, in_port, buffer_id, raw_data)
                return

            modified = False
            for rr in dns_resp.rr:
                if rr.rtype == QTYPE.A and isinstance(rr.rdata, A):
                    real_ip = str(rr.rdata)
                    if real_ip in self.app.real_to_virtual:
                        virt_ip = self.app.real_to_virtual[real_ip]
                        rr.rdata = A(virt_ip)
                        modified = True
                        self.app.logger.info(f"[dns] {real_ip} → {virt_ip}")

            if not modified:
                self._forward_original(datapath, in_port, buffer_id, raw_data)
                return

            new_dns_data = dns_resp.pack()
            new_pkt = self._build_packet(eth, ipv4_pkt, udp_pkt, new_dns_data)
            new_pkt.serialize()

            client_ip = ipv4_pkt.dst
            dpid, port = self.app.topology.get_host_location_by_ip(client_ip)
            if dpid and port:
                dp = self.app.topology.get_datapath(dpid)
                if dp:
                    actions = [dp.ofproto_parser.OFPActionOutput(port)]
                    out = dp.ofproto_parser.OFPPacketOut(
                        datapath=dp,
                        buffer_id=dp.ofproto.OFP_NO_BUFFER,
                        in_port=dp.ofproto.OFPP_CONTROLLER,
                        actions=actions,
                        data=new_pkt.data
                    )
                    dp.send_msg(out)
                    self.app.logger.info(f"[dns] direct to client {client_ip}")
                    return

            self.app.logger.warning("[dns] client unknown - fallback")
            self._forward_original(datapath, in_port, buffer_id, new_pkt.data)

        except Exception as e:
            self.app.logger.error(f"[dns] error: {e}")
            self._forward_original(datapath, in_port, buffer_id, raw_data)

    def _build_packet(self, eth, ipv4_pkt, udp_pkt, dns_data):
        p = packet.Packet()
        p.add_protocol(ethernet.ethernet(dst=eth.dst, src=eth.src, ethertype=eth.ethertype))
        p.add_protocol(ipv4.ipv4(
            version=ipv4_pkt.version,
            header_length=ipv4_pkt.header_length,
            tos=ipv4_pkt.tos,
            total_length=20 + 8 + len(dns_data),
            identification=ipv4_pkt.identification,
            flags=ipv4_pkt.flags,
            offset=ipv4_pkt.offset,
            ttl=ipv4_pkt.ttl,
            proto=ipv4_pkt.proto,
            csum=0,
            src=ipv4_pkt.src,
            dst=ipv4_pkt.dst,
            option=ipv4_pkt.option
        ))
        p.add_protocol(udp.udp(
            src_port=udp_pkt.src_port,
            dst_port=udp_pkt.dst_port,
            total_length=8 + len(dns_data),
            csum=0
        ))
        p.add_protocol(dns_data)
        return p

    def _forward_original(self, datapath, in_port, buffer_id, data):
        actions = [datapath.ofproto_parser.OFPActionOutput(datapath.ofproto.OFPP_NORMAL)]
        out = datapath.ofproto_parser.OFPPacketOut(
            datapath=datapath,
            buffer_id=buffer_id if buffer_id != datapath.ofproto.OFP_NO_BUFFER else datapath.ofproto.OFP_NO_BUFFER,
            in_port=in_port,
            actions=actions,
            data=data if buffer_id == datapath.ofproto.OFP_NO_BUFFER else None
        )
        datapath.send_msg(out)

        level=logging.INFO,
        format='%(asctime)s | %(levelname)-8s | %(message)s',

def ip_to_int(ip_str):
    """Convert IP string to integer."""
    return struct.unpack("!I", socket.inet_aton(ip_str))[0]

def int_to_ip(ip_int):
    """Convert integer to IP string."""
    return socket.inet_ntoa(struct.pack("!I", ip_int))

from . import constants as const

class PacketInHandler:
    def __init__(self, app):
        self.app = app

    def packet_in_handler(self, ev):
        msg = ev.msg
        datapath = msg.datapath
        in_port = msg.match['in_port']
        pkt = packet.Packet(msg.data)
        eth = pkt.get_protocol(ethernet.ethernet)

        if eth.ethertype == const.ETH_TYPE_LLDP:
            return

        pkt_type = self._packet_type(pkt)
        self.app.logger.info(f"\n{'='*80}")
        self.app.logger.info(f"📥 PACKET_IN | S{datapath.id}:{in_port} | {pkt_type}")
        self.app.logger.info(f"   {eth.src} → {eth.dst} | EtherType={hex(eth.ethertype)}")

        arp_pkt = pkt.get_protocol(arp.arp)
        if arp_pkt:
            self.app.arp_handler.handle_arp(datapath, in_port, pkt, eth, arp_pkt, msg.buffer_id)
            return

        ipv4_pkt = pkt.get_protocol(ipv4.ipv4)
        if ipv4_pkt:
            self.app.ipv4_handler.handle_ipv4(datapath, in_port, pkt, eth, ipv4_pkt, msg.buffer_id, msg.data)
            return

        self.app.logger.warning(f"[packet] unknown type {hex(eth.ethertype)} - flood locally")
        self._flood_locally(datapath, in_port, msg.buffer_id, msg.data)

    def _packet_type(self, pkt):
        if pkt.get_protocol(arp.arp):
            return "ARP"
        ipv4_pkt = pkt.get_protocol(ipv4.ipv4)
        if ipv4_pkt:
            proto = {1:'ICMP',6:'TCP',17:'UDP'}.get(ipv4_pkt.proto, str(ipv4_pkt.proto))
            return f"IPv4/{proto}"
        return "UNKNOWN"

    def _flood_locally(self, datapath, in_port, buffer_id, data):
        if datapath.id not in self.app.topology.access_ports:
            return
        ports = [p for p in self.app.topology.access_ports[datapath.id] if p != in_port]
        for port in ports:
            actions = [datapath.ofproto_parser.OFPActionOutput(port)]
            out = datapath.ofproto_parser.OFPPacketOut(
                datapath=datapath,
                buffer_id=buffer_id if buffer_id != datapath.ofproto.OFP_NO_BUFFER else datapath.ofproto.OFP_NO_BUFFER,
                in_port=in_port,
                actions=actions,
                data=data if buffer_id == datapath.ofproto.OFP_NO_BUFFER else None
            )
            datapath.send_msg(out)

from ryu.topology.api import get_switch, get_link
from . import constants as const

class TopologyManager:
    """Manages network topology discovery and host tracking."""

    def __init__(self, app):
        self.app = app
        self.link_to_port = {}      # {(src_dpid,dst_dpid):(src_port,dst_port)}
        self.access_table = {}       # {(sw, port): (ip, mac)}
        self.switch_port_table = {}  # {dpid: set(port_num)}
        self.access_ports = {}       # {dpid: set(port_num)} ports to hosts
        self.interior_ports = {}     # {dpid: set(port_num)} ports to switches
        self.switches = []            # list of dpids
        self.host_table = {}          # {real_ip: {'mac': mac, 'dpid': dpid, 'port': port}}
        self.mac_to_port = {}         # fallback L2: {dpid: {mac: port}}
        self.datapath_list = {}       # {dpid: datapath}

    def register_datapath(self, dpid, datapath):
        self.datapath_list[dpid] = datapath

    def unregister_datapath(self, dpid):
        self.datapath_list.pop(dpid, None)

    def get_datapath(self, dpid):
        return self.datapath_list.get(dpid)

    def discover(self):
        """Discover topology and update port classifications."""
        self.app.logger.info("[topology] Discovering network topology...")
        switch_list = get_switch(self.app, None)
        link_list = get_link(self.app, None)

        self._create_port_map(switch_list)
        self.switches = [sw.dp.id for sw in switch_list]
        self._create_interior_links(link_list)
        self._create_access_ports()
        self.app.logger.info(f"[topology] ✓ Discovered {len(self.switches)} switches, {len(self.link_to_port)} links")

    def _create_port_map(self, switch_list):
        for sw in switch_list:
            dpid = sw.dp.id
            self.switch_port_table.setdefault(dpid, set())
            self.interior_ports.setdefault(dpid, set())
            self.access_ports.setdefault(dpid, set())
            for port in sw.ports:
                if port.port_no in [const.OFPP_LOCAL, const.OFPP_CONTROLLER]:
                    continue
                self.switch_port_table[dpid].add(port.port_no)

    def _create_interior_links(self, link_list):
        self.link_to_port.clear()
        for link in link_list:
            src, dst = link.src, link.dst
            if src.port_no in [const.OFPP_LOCAL, const.OFPP_CONTROLLER] or \
               dst.port_no in [const.OFPP_LOCAL, const.OFPP_CONTROLLER]:
                continue
            self.link_to_port[(src.dpid, dst.dpid)] = (src.port_no, dst.port_no)
            if src.dpid in self.switches:
                self.interior_ports[src.dpid].add(src.port_no)
            if dst.dpid in self.switches:
                self.interior_ports[dst.dpid].add(dst.port_no)

    def _create_access_ports(self):
        for sw in self.switch_port_table:
            all_ports = self.switch_port_table[sw]
            interior = self.interior_ports.get(sw, set())
            self.access_ports[sw] = all_ports - interior - {const.OFPP_LOCAL, const.OFPP_CONTROLLER}

    def register_host(self, dpid, in_port, ip, mac):
        """Register host if port is an access port."""
        if dpid in self.access_ports and in_port in self.access_ports[dpid]:
            if ip in self.app.real_to_virtual:
                self.host_table[ip] = {'mac': mac, 'dpid': dpid, 'port': in_port}
                virt = self.app.real_to_virtual.get(ip, 'N/A')
                self.app.logger.info(f"\n[host] ✓ NEW HOST: {ip} → VIRTUAL: {virt} at S{dpid}:p{in_port} MAC:{mac}")

            prev = self.access_table.get((dpid, in_port))
            self.access_table[(dpid, in_port)] = (ip, mac)
            if prev != (ip, mac):
                if prev:

from ryu.lib.packet import packet, ethernet, arp
from . import constants as const

class ARPHandler:
    def __init__(self, app):
        self.app = app

    def handle_arp(self, datapath, in_port, pkt, eth, arp_pkt, buffer_id):
        op = "REQUEST" if arp_pkt.opcode == arp.ARP_REQUEST else "REPLY"
        self.app.logger.info(f"[arp] {op}: {arp_pkt.src_ip} ({arp_pkt.src_mac}) → {arp_pkt.dst_ip}")
        self.app.topology.register_host(datapath.id, in_port, arp_pkt.src_ip, arp_pkt.src_mac)

        if arp_pkt.opcode == arp.ARP_REQUEST:
            self._handle_request(datapath, in_port, pkt, eth, arp_pkt, buffer_id)
        else:
            self._handle_reply(datapath, in_port, pkt, eth, arp_pkt, buffer_id)

    def _handle_request(self, datapath, in_port, pkt, eth, arp_pkt, buffer_id):
        target = arp_pkt.dst_ip
        sender_ip = arp_pkt.src_ip
        sender_mac = arp_pkt.src_mac

        # DNS server
        if target == self.app.dns_ip or sender_ip == self.app.dns_ip:
            self.app.logger.info("[arp] DNS ARP - flood all hosts")
            self._flood_to_all_hosts(pkt.data, datapath.id, in_port)
            return

        # Virtual IP request
        if target in self.app.virtual_to_real:
            real_target = self.app.virtual_to_real[target]
            self.app.logger.info(f"[arp] virtual {target} → {real_target}")
            new_pkt = self._build_request(sender_mac, sender_ip, real_target)
            if real_target not in self.app.topology.host_table:
                self.app.logger.info("[arp] target unknown - flood all hosts")
                self._flood_to_all_hosts(new_pkt.data, datapath.id, in_port)
            else:
                loc = self.app.topology.host_table[real_target]
                self.app.logger.info(f"[arp] target known at S{loc['dpid']}:{loc['port']} - direct")
                self._send_to_host(new_pkt.data, loc['dpid'], loc['port'])
            return

        # Normal ARP: flood locally
        self.app.logger.info("[arp] normal ARP - flood locally")
        self._flood_locally(datapath, in_port, buffer_id, pkt.data)

    def _handle_reply(self, datapath, in_port, pkt, eth, arp_pkt, buffer_id):
        sender_ip = arp_pkt.src_ip
        sender_mac = arp_pkt.src_mac
        target_ip = arp_pkt.dst_ip

        if sender_ip == self.app.dns_ip or target_ip == self.app.dns_ip:
            self.app.logger.info("[arp] DNS reply")
            self.app.topology.register_host(datapath.id, in_port, sender_ip, sender_mac)

            # Find the location of the target (the other party)
            loc_dpid, loc_port = self.app.topology.get_host_location_by_ip(target_ip)
            if loc_dpid and loc_port:
                self.app.logger.info(f"[arp] Direct delivery to {target_ip} at S{loc_dpid}:p{loc_port}")
                self._send_to_host(pkt.data, loc_dpid, loc_port)
            else:
                self.app.logger.info("[arp] Target location unknown - flood locally")
                self._flood_locally(datapath, in_port, buffer_id, pkt.data)
            return

        if target_ip in self.app.real_to_virtual:
            virt_target = self.app.real_to_virtual[target_ip]
            reply_sender = self.app.real_to_virtual.get(sender_ip, sender_ip)
            new_pkt = self._build_reply(sender_mac, reply_sender, arp_pkt.dst_mac, virt_target)
            if virt_target in self.app.virtual_to_real:
                real_req = self.app.virtual_to_real[virt_target]
                loc = self.app.topology.get_host_location_by_ip(real_req)
                if loc[0]:
                    self._send_to_host(new_pkt.data, loc[0], loc[1])
                    return
            self.app.logger.warning("[arp] requester unknown - flood locally")
            self._flood_locally(datapath, in_port, datapath.ofproto.OFP_NO_BUFFER, new_pkt.data)
            return

        self.app.logger.info("[arp] normal reply - flood locally")
        self._flood_locally(datapath, in_port, buffer_id, pkt.data)

    def _build_request(self, src_mac, src_ip, target_ip):
        p = packet.Packet()
        p.add_protocol(ethernet.ethernet(dst='ff:ff:ff:ff:ff:ff', src=src_mac, ethertype=const.ETH_TYPE_ARP))
        p.add_protocol(arp.arp(opcode=arp.ARP_REQUEST, src_mac=src_mac, src_ip=src_ip,
                                dst_mac='ff:ff:ff:ff:ff:ff', dst_ip=target_ip))
        p.serialize()
        return p

    def _build_reply(self, src_mac, src_ip, dst_mac, dst_ip):
        p = packet.Packet()
        p.add_protocol(ethernet.ethernet(dst=dst_mac, src=src_mac, ethertype=const.ETH_TYPE_ARP))
        p.add_protocol(arp.arp(opcode=arp.ARP_REPLY, src_mac=src_mac, src_ip=src_ip,
                                dst_mac=dst_mac, dst_ip=dst_ip))
        p.serialize()
        return p

    def _flood_locally(self, datapath, in_port, buffer_id, data):
        ports = self.app.topology.access_ports.get(datapath.id, [])
        for p in ports:
            if p == in_port: continue
            actions = [datapath.ofproto_parser.OFPActionOutput(p)]
            out = datapath.ofproto_parser.OFPPacketOut(
                datapath=datapath,
                buffer_id=buffer_id if buffer_id != datapath.ofproto.OFP_NO_BUFFER else datapath.ofproto.OFP_NO_BUFFER,
                in_port=in_port,
                actions=actions,
                data=data if buffer_id == datapath.ofproto.OFP_NO_BUFFER else None
            )
            datapath.send_msg(out)

    def _flood_to_all_hosts(self, data, exclude_dpid, exclude_port):
        for dpid, ports in self.app.topology.access_ports.items():

                out = dp.ofproto_parser.OFPPacketOut(

                    buffer_id=dp.ofproto.OFP_NO_BUFFER,
                    in_port=dp.ofproto.OFPP_CONTROLLER,

                dp.send_msg(out)

    def _send_to_host(self, data, dpid, port):
        dp = self.app.topology.get_datapath(dpid)
        if dp:
            actions = [dp.ofproto_parser.OFPActionOutput(port)]
            out = dp.ofproto_parser.OFPPacketOut(
                datapath=dp,
                buffer_id=dp.ofproto.OFP_NO_BUFFER,
                in_port=dp.ofproto.OFPP_CONTROLLER,
                actions=actions,
                data=data
            )
            dp.send_msg(out)
